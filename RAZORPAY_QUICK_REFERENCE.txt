2. CREATE PAYMENT ORDER (payments/razorpay.ts)

   createPaymentOrder(bookingId) {
     // Validates: booking.customerId === userId
     // Validates: booking.bookingStatus === "awaiting_payment"
     // Validates: booking.paymentMethod === "after_service"
     
     // Get amount from Firestore (NEVER trust client)
     const amount = booking.pricing?.total || booking.finalAmount;
     
     // Create Razorpay order
     const order = await razorpay.orders.create({
       amount: Math.round(amount * 100), // paise
       currency: 'INR',
       receipt: booking.bookingNumber,
       notes: {
         bookingId,
         customerId: userId,
         technicianId: booking.technicianId,
         paymentMethod: 'after_service'
       }
     });
     
     // Store in razorpayOrders collection
     await createRazorpayOrderDoc(order.id, amount, { bookingId });
     
     // Update booking with order ID
     await bookingRef.update({
       'payment.razorpayOrderId': order.id,
       'payment.status': 'processing',
       'paymentStatus': 'processing'
     });
     
     return { orderId: order.id, amount, currency: 'INR' };
   }

3. VERIFY PAYMENT (payments/razorpay.ts)
   ────────────────────────────────────
   verifyPayment(bookingId, razorpayOrderId, razorpayPaymentId, razorpaySignature) {
     // Validates: booking.customerId === userId
     // Validates: booking.bookingStatus === "awaiting_payment"
     // Validates: signature is valid
     
     // Fetch payment from Razorpay
     const payment = await razorpay.payments.fetch(razorpayPaymentId);
     const amount = payment.amount / 100; // rupees
     
     // Verify amount matches
     if (Math.abs(amount - bookingTotal) > 0.01) throw error;
     
     // TRANSACTION: Update booking + credit wallet atomically
     await db.runTransaction(async (transaction) => {
       // Check if already paid (race condition protection)
       const currentBooking = await transaction.get(bookingRef);
       if (currentBooking.payment?.status === 'paid') return; // idempotent
       
       // Update booking
       transaction.update(bookingRef, {
         'payment.status': 'paid',
         'payment.razorpayPaymentId': razorpayPaymentId,
         'payment.amountPaid': amount,
         'paymentStatus': 'paid',
         'paidAt': admin.firestore.FieldValue.serverTimestamp()
       });
       
       // Credit technician wallet
       const techWalletRef = db.collection('technician_wallets')
         .doc(currentBooking.technicianId);
       
       const technicianAmount = amount - (currentBooking.pricing?.platformFee || 0);
       
       transaction.update(techWalletRef, {
         availableBalance: admin.firestore.FieldValue.increment(technicianAmount),
         updatedAt: admin.firestore.FieldValue.serverTimestamp()
       });
       
       // Log wallet transaction
       transaction.set(techWalletRef.collection('transactions').doc(), {
         type: 'credit',
         source: 'booking_payment',
         amount: technicianAmount,
         referenceId: bookingId,
         paymentId: razorpayPaymentId,
         createdAt: admin.firestore.FieldValue.serverTimestamp()
       });
     });
     
     return { success: true };
   }

3. HANDLE PAYMENT SUCCESS (payment_screen.dart)

   _handlePaymentSuccess(PaymentSuccessResponse response) async {
     // Step 3: Verify payment on backend
     final verifyResponse = await FunctionsService().verifyPayment(
       bookingId: widget.bookingId,
       razorpayOrderId: response.orderId!,
       razorpayPaymentId: response.paymentId!,
       razorpaySignature: response.signature!,
     );
     
     if (verifyResponse['success']) {
       // Payment verified, booking status = "paid"
       // Technician wallet credited
       // Customer can now rate service
       Navigator.pop(context, true);
     }
   }
